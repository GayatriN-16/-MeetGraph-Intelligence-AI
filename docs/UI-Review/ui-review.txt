import React, { useMemo, useState } from 'react';

import type {
  NavigationTab,
  Meeting,
  Decision,
  Task,
  User,
} from './types';

import {
  INITIAL_MEETINGS,
  INITIAL_DECISIONS,
  INITIAL_ACTION_ITEMS,
  TIMELINE_CONFLICT,
  EXTRACTION_PIPELINE,
} from './data/mockData';

import { Sidebar } from './components/Sidebar';
import { TopHeader } from './components/TopHeader';
import { Dashboard } from './components/Dashboard';
import { MeetingsLibrary } from './components/MeetingsLibrary';
import { ReactFlowKnowledgeGraph } from './components/ReactFlowKnowledgeGraph';
import { TasksAndDecisions } from './components/TasksAndDecisions';
import { AskAI } from './components/AskAI';
import { ArchitecturePipeline } from './components/ArchitecturePipeline';
import { ExternalIntegrations } from './components/ExternalIntegrations';
import { AdminAuditLogs } from './components/AdminAuditLogs';
import { ProjectsView } from './components/ProjectsView';
import { SettingsView } from './components/SettingsView';
import { NotificationsView } from './components/NotificationsView';

import { MeetingDetailModal } from './components/MeetingDetailModal';
import { ConflictEvidenceModal } from './components/ConflictEvidenceModal';
import { UploadMeetingModal } from './components/UploadMeetingModal';
import { AuthModal } from './components/AuthModal';

const DEFAULT_USER: User = {
  id: 'usr-admin-1',
  email: 'alex.vance@meetgraph.ai',
  name: 'Alex Vance',
  role: 'Admin',
  avatarInitials: 'AV',
};

const DEFAULT_PROJECT = 'Acme Engineering - Capstone v2.4';

export default function App() {
  // Application state
  const [currentUser, setCurrentUser] = useState<User>(DEFAULT_USER);
  const [currentTab, setCurrentTab] =
    useState<NavigationTab>('dashboard');

  const [selectedProject, setSelectedProject] =
    useState<string>(DEFAULT_PROJECT);

  const [searchQuery, setSearchQuery] = useState('');

  // Meeting and intelligence data
  const [meetings, setMeetings] =
    useState<Meeting[]>(INITIAL_MEETINGS);

  const [decisions, setDecisions] =
    useState<Decision[]>(INITIAL_DECISIONS);

  const [actionItems, setActionItems] =
    useState<Task[]>(INITIAL_ACTION_ITEMS);

  const timelineConflict = TIMELINE_CONFLICT;

  // Modal state
  const [selectedMeeting, setSelectedMeeting] =
    useState<Meeting | null>(null);

  const [showConflictModal, setShowConflictModal] =
    useState(false);

  const [showUploadModal, setShowUploadModal] =
    useState(false);

  const [showAuthModal, setShowAuthModal] =
    useState(false);

  const [askAIPrompt, setAskAIPrompt] = useState('');

  // Filter meetings using the global search query
  const filteredMeetings = useMemo(() => {
    const query = searchQuery.trim().toLowerCase();

    if (!query) {
      return meetings;
    }

    return meetings.filter((meeting) => {
      const searchableText = [
        meeting.title,
        meeting.project,
        meeting.summary,
        meeting.audioFileName ?? '',
        ...meeting.speakers.map((speaker) => speaker.name),
        ...meeting.keyTakeaways,
        ...(meeting.keyDiscussionPoints ?? []),
        ...meeting.transcript.map((segment) => segment.text),
      ]
        .join(' ')
        .toLowerCase();

      return searchableText.includes(query);
    });
  }, [meetings, searchQuery]);

  // Toggle task completion
  const handleToggleActionItem = (id: string) => {
    setActionItems((previousItems) =>
      previousItems.map((item) =>
        item.id === id
          ? { ...item, completed: !item.completed }
          : item,
      ),
    );
  };

  // Add a new task
  const handleAddActionItem = (newItem: Task) => {
    setActionItems((previousItems) => [
      newItem,
      ...previousItems,
    ]);
  };

  // Handle meeting upload
  const handleMeetingUploaded = (newMeeting: Meeting) => {
    setMeetings((previousMeetings) => [
      newMeeting,
      ...previousMeetings,
    ]);

    setShowUploadModal(false);
    setSelectedMeeting(newMeeting);
  };

  // Open AI assistant with meeting context
  const handleAskAIAboutMeeting = (meetingTitle: string) => {
    setSelectedMeeting(null);

    setAskAIPrompt(
      `Summarize key decisions and open questions from ${meetingTitle}`,
    );

    setCurrentTab('ask-ai');
  };

  // Resolve the current demonstration conflict
  const handleResolveConflict = () => {
    setShowConflictModal(false);

    setActionItems((previousItems) =>
      previousItems.map((item) =>
        item.id === 'act-4'
          ? {
              ...item,
              completed: true,
              taskDescription:
                '[RESOLVED] Synchronized Client Review target to Oct 28',
            }
          : item,
      ),
    );

    setCurrentTab('tasks-decisions');
  };

  // Global meeting search
  const handleSearch = (query: string) => {
    setSearchQuery(query);

    if (query.trim()) {
      setCurrentTab('meetings');
    }
  };

  // Open a meeting from the knowledge graph
  const handleSelectMeetingFromGraph = (title: string) => {
    const normalizedTitle = title.trim().toLowerCase();

    if (!normalizedTitle) {
      return;
    }

    const matchingMeeting = meetings.find((meeting) =>
      meeting.title.toLowerCase().includes(normalizedTitle),
    );

    if (matchingMeeting) {
      setSelectedMeeting(matchingMeeting);
    }
  };

  return (
    <div className="min-h-screen w-full bg-[#f8f9ff] font-sans text-slate-800 selection:bg-violet-100 selection:text-violet-950">
      <div className="flex min-h-screen w-full">

        {/* Sidebar */}
        <Sidebar
          currentTab={currentTab}
          currentUser={currentUser}
          onSelectTab={setCurrentTab}
          openUploadModal={() => setShowUploadModal(true)}
        />

        {/* Main application area */}
        <div className="flex min-h-screen min-w-0 flex-1 flex-col">

          {/* Header */}
          <TopHeader
            currentUser={currentUser}
            onOpenAuthModal={() => setShowAuthModal(true)}
            onSearch={handleSearch}
            openUploadModal={() => setShowUploadModal(true)}
            onSelectTab={setCurrentTab}
            selectedProject={selectedProject}
            setSelectedProject={setSelectedProject}
          />

          {/* Page content */}
          <main className="min-w-0 flex-1 overflow-x-hidden pb-8">

            {currentTab === 'dashboard' && (
              <Dashboard
                meetings={meetings}
                decisions={decisions}
                actionItems={actionItems}
                timelineConflict={timelineConflict}
                extractionPipeline={EXTRACTION_PIPELINE}
                onSelectMeeting={setSelectedMeeting}
                onOpenConflictModal={() =>
                  setShowConflictModal(true)
                }
                onToggleActionItem={handleToggleActionItem}
                onSelectTab={setCurrentTab}
                openUploadModal={() =>
                  setShowUploadModal(true)
                }
              />
            )}

            {currentTab === 'meetings' && (
              <MeetingsLibrary
                meetings={filteredMeetings}
                onSelectMeeting={setSelectedMeeting}
                openUploadModal={() =>
                  setShowUploadModal(true)
                }
              />
            )}

            {currentTab === 'knowledge-graph' && (
              <ReactFlowKnowledgeGraph
                onSelectMeetingTitle={
                  handleSelectMeetingFromGraph
                }
              />
            )}

            {currentTab === 'tasks-decisions' && (
              <TasksAndDecisions
                decisions={decisions}
                actionItems={actionItems}
                meetings={meetings}
                onToggleActionItem={handleToggleActionItem}
                onAddActionItem={handleAddActionItem}
                onSelectMeeting={setSelectedMeeting}
              />
            )}

            {currentTab === 'ask-ai' && (
              <AskAI initialPrompt={askAIPrompt} />
            )}

            {currentTab === 'notifications' && (
              <NotificationsView />
            )}

            {currentTab === 'architecture' && (
              <ArchitecturePipeline />
            )}

            {currentTab === 'integrations' && (
              <ExternalIntegrations />
            )}

            {currentTab === 'admin-audit' && (
              <AdminAuditLogs />
            )}

            {currentTab === 'projects' && (
              <ProjectsView />
            )}

            {currentTab === 'settings' && (
              <SettingsView />
            )}

          </main>
        </div>
      </div>

      {/* Meeting details */}
      {selectedMeeting && (
        <MeetingDetailModal
          meeting={selectedMeeting}
          onClose={() => setSelectedMeeting(null)}
          onAskAIAboutMeeting={handleAskAIAboutMeeting}
        />
      )}

      {/* Conflict evidence */}
      {showConflictModal && (
        <ConflictEvidenceModal
          conflict={timelineConflict}
          onClose={() => setShowConflictModal(false)}
          onResolve={handleResolveConflict}
        />
      )}

      {/* Meeting upload */}
      {showUploadModal && (
        <UploadMeetingModal
          onClose={() => setShowUploadModal(false)}
          onMeetingUploaded={handleMeetingUploaded}
        />
      )}

      {/* Authentication */}
      {showAuthModal && (
        <AuthModal
          currentUser={currentUser}
          onClose={() => setShowAuthModal(false)}
          onUserChanged={setCurrentUser}
        />
      )}
    </div>
  );
}

const panelClass =
  'rounded-2xl border border-slate-200/80 bg-white shadow-sm shadow-slate-200/40';

const sectionTitleClass = 'text-sm font-semibold text-slate-900';

const formatDate = (value: string) => {
  if (!value) return 'Date unavailable';

  const parsed = new Date(value);

  return Number.isNaN(parsed.getTime())
    ? value
    : parsed.toLocaleDateString('en-US', {
        month: 'short',
        day: 'numeric',
        year: 'numeric',
      });
};

const StatusPill: React.FC<{
  children: React.ReactNode;
  tone?: 'green' | 'amber' | 'rose' | 'violet' | 'slate';
}> = ({ children, tone = 'slate' }) => {
  const tones = {
    green: 'bg-emerald-50 text-emerald-700 ring-emerald-600/15',
    amber: 'bg-amber-50 text-amber-700 ring-amber-600/15',
    rose: 'bg-rose-50 text-rose-700 ring-rose-600/15',
    violet: 'bg-violet-50 text-violet-700 ring-violet-600/15',
    slate: 'bg-slate-100 text-slate-600 ring-slate-500/15',
  };

  return (
    <span
      className={`inline-flex items-center rounded-full px-2.5 py-1 text-[11px] font-medium ring-1 ring-inset ${tones[tone]}`}
    >
      {children}
    </span>
  );
};

                    Recorded decisions and their source meetings

                </div>
              </div>
              <button
                type="button"
                onClick={() => onSelectTab('tasks-decisions')}
                className="text-xs font-semibold text-violet-700 hover:text-violet-900"

                All decisions

              <p className="py-8 text-center text-sm text-slate-500">
                No decisions have been recorded yet.
              </p>
            ) : (
              <div className="mt-4 space-y-3">
                {decisions.slice(0, 3).map((decision) => {
                  const sourceMeeting = meetings.find(
                    (meeting) => meeting.id === decision.sourceMeetingId,
                  );

                  return (
                    <article

                          {decision.title}

                        <StatusPill

                            decision.status === 'Verified' ||
                            decision.status === 'User Confirmed'
                              ? 'green'
                              : decision.status === 'Needs Review'
                                ? 'amber'
                                : 'violet'
                          }
                        >
                          {decision.status}
                        </StatusPill>

                      className={`flex w-full items-center justify-between gap-3 rounded-xl px-3 py-2.5 text-left text-xs transition ${
                        active
                          ? 'bg-violet-50 font-semibold text-violet-700'
                          : 'text-slate-600 hover:bg-slate-50'
                      }`}
                    >
                      <span>{project}</span>
                      {active && (
                        <Check className="h-4 w-4 shrink-0 text-violet-600" />
                      )}
                    </button>
                  );
                })}
              </div>
            </>
          )}
        </div>

        {/* Notifications */}
        <button
          type="button"
          onClick={() => onSelectTab('notifications')}
          title="Notifications"
          aria-label={`Notifications${
            unreadCount > 0 ? `, ${unreadCount} unread` : ''
          }`}
          className="relative flex h-10 w-10 shrink-0 items-center justify-center rounded-xl border border-transparent text-slate-500 transition hover:border-slate-200 hover:bg-slate-50 hover:text-violet-700"
        >
          <Bell className="h-[18px] w-[18px]" />
          {unreadCount > 0 && (
            <span className="absolute right-2 top-2 h-2 w-2 rounded-full bg-rose-500 ring-2 ring-white" />
          )}
        </button>

        {/* Account */}
        <button
          type="button"
          onClick={onOpenAuthModal}
          aria-label={`Account: ${currentUser.name}`}
          className="flex shrink-0 items-center gap-2.5 rounded-xl border border-slate-200 bg-white px-2.5 py-1.5 text-left transition hover:border-violet-200 hover:bg-violet-50/30"
        >
          <span className="flex h-8 w-8 items-center justify-center rounded-full bg-gradient-to-br from-violet-100 to-indigo-100 text-xs font-bold text-violet-700 ring-1 ring-inset ring-violet-200/70">
            {currentUser.avatarInitials}
          </span>

          <span className="hidden min-w-0 sm:block">
            <span className="block max-w-28 truncate text-xs font-semibold text-slate-800">
              {currentUser.name}
            </span>
            <span className="mt-0.5 flex items-center gap-1 text-[10px] text-slate-500">
              <ShieldCheck className="h-3 w-3 text-teal-600" />
              {currentUser.role}
            </span>
          </span>
        </button>

        {/* Quick upload */}
        <button
          type="button"
          onClick={openUploadModal}
          className="hidden items-center gap-2 rounded-xl bg-violet-600 px-3.5 py-2.5 text-xs font-semibold text-white shadow-sm shadow-violet-200 transition hover:bg-violet-700 focus:outline-none focus:ring-4 focus:ring-violet-500/20 xl:flex"
        >
          <UploadCloud className="h-4 w-4" />
          Upload meeting
        </button>

        {/* Ask AI shortcut */}
        <button
          type="button"
          onClick={() => onSelectTab('ask-ai')}
          aria-label="Open Ask AI"
          title="Ask AI"
          className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl border border-violet-100 bg-violet-50 text-violet-700 transition hover:border-violet-200 hover:bg-violet-100"
        >
          <Sparkles className="h-[18px] w-[18px]" />
        </button>
      </div>
    </header>
  );
};
